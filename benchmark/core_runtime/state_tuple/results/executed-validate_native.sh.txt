#!/bin/bash
set -euo pipefail
cd $WORKSPACE
source deps/env.sh
unset PYTHONPATH ENVPOOL_ASSETS_PATH
export PYTHONDONTWRITEBYTECODE=1
W=$PWD/envpool-core-state-tuple-work
S=$PWD/envpool-core-state-tuple
.venv/bin/python envpool-core-work/tools/build_aux.py --source "$S" --out "$W/build/native" --gtest "$S"/envpool/core/*_test.cc "$S/envpool/dummy/dummy_envpool_test.cc" > "$W/logs/build-native.log" 2>&1
timeout 240 "$W/build/native" --gtest_output=xml:"$W/results/native.xml" > "$W/logs/native.log" 2>&1
tail -n 4 "$W/logs/native.log" >> envpool-core-overnight-work/logs/progress.log
for san in address,undefined thread; do
 tag=${san%%,*}
 .venv/bin/python envpool-core-work/tools/build_aux.py --source "$S" --out "$W/build/$tag" --gtest --sanitize "$san" "$S"/envpool/core/*_test.cc "$S/envpool/dummy/dummy_envpool_test.cc" > "$W/logs/build-$tag.log" 2>&1
 ASAN_OPTIONS=detect_leaks=0 TSAN_OPTIONS=halt_on_error=1 timeout 240 "$W/build/$tag" --gtest_filter='*-DummyEnvPoolTest.SinglePlayer:DummyEnvPoolTest.MultiPlayers' --gtest_repeat=3 --gtest_output=xml:"$W/results/$tag.xml" > "$W/logs/$tag.log" 2>&1
 printf 'State tuple %s: ' "$tag" >> envpool-core-overnight-work/logs/progress.log
 tail -n 1 "$W/logs/$tag.log" >> envpool-core-overnight-work/logs/progress.log
done
