#!/bin/bash
set -euo pipefail
cd $WORKSPACE
source deps/env.sh
unset PYTHONPATH ENVPOOL_ASSETS_PATH
export PYTHONDONTWRITEBYTECODE=1 PYTHONHASHSEED=0 OPENBLAS_NUM_THREADS=1 OMP_NUM_THREADS=1 MKL_NUM_THREADS=1 NUMBA_NUM_THREADS=1
S=$PWD/envpool-core-ppo-profile/benchmark/core_runtime
W=$PWD/envpool-core-ppo-profile-work
for runtime in original retained; do
 if [[ $runtime = original ]]; then package=$PWD/.venv/lib/python3.12/site-packages/envpool; else package=$PWD/envpool-core-state-tuple-work/runtime-candidate/envpool; fi
 hash=$(sha256sum "$package/classic_control/classic_control_envpool.so" | cut -d' ' -f1)
 for mode in plain profile; do
  if [[ $mode = plain ]]; then script=$S/experiments/bench_ppo.py; else script=$S/ppo_phase/profile_ppo.py; fi
  .rl-venv/bin/python "$script" --package "$package" --label pilot --output "$W/results/pilot-$runtime-$mode.json" --iterations 2 --warmup 2 --native-sha256 "$hash" > "$W/logs/pilot-$runtime-$mode.log" 2>&1
 done
done
.venv/bin/python - <<'PY'
from pathlib import Path
import json
w=Path('envpool-core-ppo-profile-work/results');rows=[json.loads(p.read_text()) for p in sorted(w.glob('pilot-*-*.json'))];assert len(rows)==4;assert len({d['semantic_sha256'] for d in rows})==1
for d in rows:
 assert d['budget']['env_steps']==5120 and d['budget']['optimizer_steps']==160
 if 'phase_profile' in d:
  p=d['phase_profile'];assert p['validation']['valid'] and p['validation']['expected_updates']==2 and p['phases']['env_step']['calls']==256
(w/'pilot-validation.json').write_text(json.dumps({'passed':True,'fresh_processes':4,'updates_per_process':2,'env_steps':5120,'optimizer_steps':160,'one_exact_semantic_fingerprint':True,'profile_counts_valid':True,'timings_not_used_for_performance_claims':True},indent=2)+'\n');print('Four pilot runs: exact semantic parity and profile counts verified')
PY
