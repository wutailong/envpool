#!/bin/bash
set -euo pipefail
cd $WORKSPACE
source deps/env.sh
unset PYTHONPATH ENVPOOL_ASSETS_PATH
export PYTHONDONTWRITEBYTECODE=1 OPENBLAS_NUM_THREADS=1 OMP_NUM_THREADS=1 MKL_NUM_THREADS=1 NUMBA_NUM_THREADS=1 PYTHONHASHSEED=0
W=$PWD/envpool-core-player-actions-work
S=$PWD/envpool-core-player-actions
BASE=$PWD/envpool-core-state-tuple-work/runtime-box-candidate
NEXT=$W/runtime-box-candidate
.venv/bin/python -m unittest discover -s "$S/benchmark/core_runtime/experiments" -p 'test_*.py' > "$W/logs/inherited-harness-tests.log" 2>&1
.venv/bin/python "$W/tools/verify_frozen.py" before
printf 'Player action index: all correctness gates passed, including float Container Box2D test-mode parity. Starting finite serialized cost trial: 160 general/Box2D + 64 Dummy + 32 interleaved samples, then 16 same-binary-control PPO trials.\n' >> envpool-core-overnight-work/logs/progress.log
.venv/bin/python "$S/benchmark/core_runtime/experiments/run_blocks.py" --python "$PWD/.venv/bin/python" --variant a="$BASE" --variant b="$BASE" --variant c="$NEXT" --variant d="$NEXT" --paired-replicates --out "$W/results/performance.jsonl" --cases "$W/results/cases.json" --blocks 8 --seconds 3 --warmup 400 --affinities default > "$W/logs/performance.log" 2>&1
.venv/bin/python "$S/benchmark/core_runtime/container_ownership/run_container_blocks.py" --python "$PWD/.venv/bin/python" --baseline-root "$BASE" --candidate-root "$NEXT" --out "$W/results/container-performance.jsonl" > "$W/logs/container-performance.log" 2>&1
.venv/bin/python "$S/benchmark/core_runtime/player_action_index/run_interleaved_blocks.py" --python "$PWD/.venv/bin/python" --baseline-root "$BASE" --candidate-root "$NEXT" --out "$W/results/interleaved-performance.jsonl" > "$W/logs/interleaved-performance.log" 2>&1
printf 'Player action index: 256 balanced throughput samples saved; running separate full-budget PPO throughput blocks.\n' >> envpool-core-overnight-work/logs/progress.log
.rl-venv/bin/python "$S/benchmark/core_runtime/player_action_index/run_paired_ppo.py" --python "$PWD/.rl-venv/bin/python" --baseline-root "$BASE" --candidate-root "$NEXT" --output "$W/results/ppo-throughput" > "$W/logs/ppo-throughput.log" 2>&1
.venv/bin/python "$W/tools/verify_frozen.py" after
printf 'Player action index: finite cost trial complete; all frozen hashes unchanged and PPO fingerprints equal.\n' >> envpool-core-overnight-work/logs/progress.log
